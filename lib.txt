function PulseUI:CreateWindow(options)

    local Window = {}

    function Window:AddTabDivider(name) end

    function Window:CreateTab(tabOptions)

        local tabName = type(tabOptions) == "table" and tabOptions.Name or tabOptions

        local Tab = Main:Tab(tabName, 0, tabName, "")

        local TabWrapper = {}

        function TabWrapper:CreateGroup(groupName, side)

            side = side == "right" and "Right" or "Left"

            Tab:Label(side, groupName)

            local Group = {}

            function Group:AddToggle(name, default, callback)

                Tab:Toggle(side, name, default, callback)

            end

            function Group:AddSlider(name, min, max, default, callback)

                Tab:Slider(side, name, default, min, max, 1, callback)

            end

            function Group:AddButton(name, callback)

                Tab:Button(side, name, callback)

            end

            function Group:AddDropdown(name, list, default, callback)

                list = list or {}

                default = tostring(default or list[1] or "")

                Tab:Dropdown(side, name, default, list, callback)

                local DropdownWrapper = {}

                function DropdownWrapper:UpdateList(newList)

                    -- Not supported in kk.lua wrapper

                end

                return DropdownWrapper

            end

            function Group:AddLabel(text)

                Tab:Label(side, text)

            end

            return Group

        end

        return TabWrapper

    end

    return Window

end

local function UniversalFireProximityPrompt(prompt, holdDuration)

    if not prompt or not prompt:IsA("ProximityPrompt") then return end

        pcall(function()

            fireproximityprompt(prompt, holdDuration)

        end)

        return

    end

    pcall(function()

        prompt:InputHoldBegin()

        if holdDuration > 0 then

            task.wait(holdDuration)

        end

        prompt:InputHoldEnd()

    end)

end

        return

    if typeof(input) == "Vector3" then

        destinationCFrame = CFrame.new(input)

    elseif typeof(input) == "CFrame" then

        destinationCFrame = input

        local targetPlayer = findPlayerByPartialName(input)

        if targetPlayer and targetPlayer.Character and targetPlayer.Character:FindFirstChild("HumanoidRootPart") then

    humanoid:ChangeState(0)

    repeat task.wait() until not LocalPlayer:GetAttribute("LastACPos")

    hrp.CFrame = destinationCFrame

    getgenv().SwimMethod = false

local function MarketHasItem(toolName)

    local market = ReplicatedStorage:FindFirstChild("MarketItems")

    if not market then return false end

    for _, item in ipairs(market:GetChildren()) do

        if item.Name == toolName and item:FindFirstChild("owner") and item.owner.Value == LocalPlayer.Name then

            return true

        end

    end

    return false

end

local function BackpackHasItem(toolName)

    for _, tool in ipairs(LocalPlayer.Backpack:GetChildren()) do

        if tool:IsA("Tool") and tool.Name == toolName then

            return true

        end

    end

    return false

end

local function ForcePutToMarket(toolName)

    local endTime = os.clock() + 12 

    while os.clock() < endTime do

        if MarketHasItem(toolName) then

            return true

        end

        pcall(function()

            ReplicatedStorage.ListWeaponRemote:FireServer(toolName, 999999)

        end)

        task.wait(0.15)

    end

    return false

end

local function ForceRetrieveFromMarket(toolName)

    local market = ReplicatedStorage:WaitForChild("MarketItems", 10)

    if not market then return false end

    local endTime = os.clock() + 12

    while os.clock() < endTime do

        for _, item in ipairs(market:GetChildren()) do

            if item.Name == toolName

                and item:FindFirstChild("owner")

                and item.owner.Value == LocalPlayer.Name

                and item:GetAttribute("SpecialId") then

                local special = item:GetAttribute("SpecialId")

                pcall(function()

                    ReplicatedStorage.BuyItemRemote:FireServer(toolName, "Remove", special)

                end)

                task.wait(0.2)

                pcall(function()

                    ReplicatedStorage.BackpackRemote:InvokeServer("Grab", toolName)

                end)

                task.wait(0.2)

                if BackpackHasItem(toolName) then

                    return true

                end

            end

        end

        task.wait(0.2)

    end

    return false

end

                ForcePutToMarket(tool.Name)

            end

        end

    end)

end

            end

        end

    end)

local function cleanup()

	if movementConnection then movementConnection:Disconnect() movementConnection=nil end

	if freezeConnection then freezeConnection:Disconnect() freezeConnection=nil end

	if bodyGyro then bodyGyro:Destroy() bodyGyro=nil end

	if animationTrack then animationTrack:Stop() animationTrack=nil end

end

local function setupMovement()

	if not humanoidRootPart then return end

	bodyGyro=Instance.new("BodyGyro")

	bodyGyro.MaxTorque=Vector3.new(math.huge,math.huge,math.huge)

	bodyGyro.P=50000

	bodyGyro.D=1000

	bodyGyro.Parent=humanoidRootPart

	movementConnection=RunService.RenderStepped:Connect(function()

		if not enhancedWalk then return end

		local camera,dir,usingKeys=workspace.CurrentCamera,Vector3.zero,false

		if UserInputService:IsKeyDown(Enum.KeyCode.W) then

			dir+=Vector3.new(camera.CFrame.LookVector.X,0,camera.CFrame.LookVector.Z)

			usingKeys=true

		end

		if UserInputService:IsKeyDown(Enum.KeyCode.S) then

			dir-=Vector3.new(camera.CFrame.LookVector.X,0,camera.CFrame.LookVector.Z)

			usingKeys=true

		end

		if UserInputService:IsKeyDown(Enum.KeyCode.A) then

			dir-=camera.CFrame.RightVector

			usingKeys=true

		end

		if UserInputService:IsKeyDown(Enum.KeyCode.D) then

			dir+=camera.CFrame.RightVector

			usingKeys=true

		end

		if not usingKeys then

			local md=humanoid.MoveDirection

			dir=Vector3.new(md.X,0,md.Z)

		end

		local moveSpeed=speed

		if UserInputService:IsKeyDown(Enum.KeyCode.LeftShift) then

			moveSpeed*=boostMultiplier

		end

		if dir.Magnitude>0 then

			dir=dir.Unit*moveSpeed

		end

		local groundedY=math.clamp(humanoidRootPart.AssemblyLinearVelocity.Y,-100,-2)

		humanoidRootPart.AssemblyLinearVelocity=

			Vector3.new(dir.X,groundedY,dir.Z)

		if dir.Magnitude>0 then

			bodyGyro.CFrame=CFrame.new(

				humanoidRootPart.Position,

				humanoidRootPart.Position+Vector3.new(dir.X,0,dir.Z)

			)

		end

		humanoidRootPart.RotVelocity=Vector3.zero

		humanoidRootPart.AssemblyAngularVelocity=Vector3.zero

		if animationTrack then

			if dir.Magnitude>0 then

				if not animationTrack.IsPlaying then animationTrack:Play() end

			else

				if animationTrack.IsPlaying then animationTrack:Stop() end

			end

		end

	end)

end

	cleanup()

	humanoidRootPart.AssemblyLinearVelocity=Vector3.zero

	local animator=humanoid:FindFirstChildWhichIsA("Animator") or Instance.new("Animator",humanoid)

	animationTrack=animator:LoadAnimation(animation)

	animationTrack.Looped=true

	task.delay(1,function()

		if freezeConnection then freezeConnection:Disconnect() freezeConnection=nil end

		if enhancedWalk then setupMovement() end

	end)

end

local function stopEnhancedWalk()

	enhancedWalk=false

	cleanup()

	getgenv().SwimMethod=false

end

local function nearSeat(hrp)

    local m = workspace:FindFirstChild("1# Map")

    if not m then return end

    local r, d

    for _, v in ipairs(m:GetDescendants()) do

        if v:IsA("Seat")  

        and not v.Disabled

        and v.CanCollide then

            if v.Position.Y < hrp.Position.Y - 8 then

                continue

            end

            local x = (v.Position - hrp.Position).Magnitude

            if not d or x < d then

                d = x

                r = v

            end

        end

    end

    if r then

        r.CFrame = r.CFrame + Vector3.new(0, 1.5, 0)

    end

    return r

end

                local camCF = camera.CFrame

                local _, yRot, _ = camCF:ToOrientation()

                local uprightRot = CFrame.Angles(0, yRot, 0)

                local flatCamCF = CFrame.new(camCF.Position) * uprightRot       

                local localInput = flatCamCF:VectorToObjectSpace(moveDir * speed)

                local worldMovement = camCF:VectorToWorldSpace(localInput)      



                head.CFrame = CFrame.new(head.Position + worldMovement) * uprightRot

                camera.CameraSubject = head

            end)

        else

            -- Cleanup

            if CFloop then CFloop:Disconnect() CFloop = nil end

            local character = LocalPlayer.Character

            if character then

                local humanoid = character:FindFirstChildOfClass("Humanoid")    

                local head = character:FindFirstChild("Head")

                if humanoid then

                    humanoid.PlatformStand = false

                    humanoid:ChangeState(Enum.HumanoidStateType.Running)        

                end

                if head then head.Anchored = false end

local function GetNearestCar(hrp)

    local CivCars = workspace:FindFirstChild("CivCars")

    if not CivCars then return nil end

    local nearestCar = nil

    local nearestDist = math.huge

    for _, car in ipairs(CivCars:GetChildren()) do

        if not car:IsA("Model") then continue end

        local seat = car:FindFirstChild("DriveSeat")

        if not seat then continue end

        if seat.Occupant then continue end

        if not car.PrimaryPart then

            local body = car:FindFirstChild("Body")

            if body and body:FindFirstChild("#Weight") then

                car.PrimaryPart = body["#Weight"]

            else

                car.PrimaryPart = seat

            end

        end

        if not car.PrimaryPart then continue end

        local dist = (car.PrimaryPart.Position - hrp.Position).Magnitude

        if dist < nearestDist then

            nearestDist = dist

            nearestCar = car

        end

    end

    return nearestCar

end

    local speed = Config.TheBronx.carflyspeed or 120

    local cam = workspace.CurrentCamera

    if not UIS.TouchEnabled then

        if UIS:IsMouseButtonPressed(Enum.UserInputType.MouseButton2) then

            bg.CFrame = CFrame.lookAt(root.Position, root.Position + camLook)

        end

        local move = Vector3.zero

        if UIS:IsKeyDown(Enum.KeyCode.W) then move += camLook end

        if UIS:IsKeyDown(Enum.KeyCode.S) then move -= camLook end

        if UIS:IsKeyDown(Enum.KeyCode.A) then move -= camRight end

        if UIS:IsKeyDown(Enum.KeyCode.D) then move += camRight end

        if move.Magnitude > 0 then

            bv.Velocity = move.Unit * speed

        else

            bv.Velocity = Vector3.zero

        end

    else

        bg.CFrame = CFrame.lookAt(root.Position, root.Position + camLook)

        local move = hum.MoveDirection

        local vertical = camLook.Y 

        local velocity =

            (move * speed) +

            (Vector3.new(0, vertical, 0) * speed)

        if velocity.Magnitude > 0 then

            bv.Velocity = velocity

        else

            bv.Velocity = Vector3.zero

        end

    end

end)

local function ScanPlayer(player)

    if not DetectAdminEnabled then return end

    if not player or player == LocalPlayer then return end

    local groupRank = player:GetAttribute("GroupRank")

    if groupRank and groupRank > 5 then 

        detectedAdmins[player] = true

        Notify("⚠ Admin Detected", player.Name .. " detected! Leaving game...", 3)

        task.wait(1)

        LocalPlayer:Kick("Bypass.cash Security | Admin Detected")

    end

end

local function ScanAllPlayers()

    Notify("Scanning", "Scanning all players...", 2)

    local foundAdmin = false

    for _, player in ipairs(Players:GetPlayers()) do

        if player ~= LocalPlayer then

            local groupRank = player:GetAttribute("GroupRank")

            if groupRank and groupRank > 5 then

                foundAdmin = true

            end

        end

        ScanPlayer(player)

        player:GetAttributeChangedSignal("GroupRank"):Connect(function()

            ScanPlayer(player)

        end)

    end

    if not foundAdmin then

        Notify("Scan Completed", "No admins detected.", 3)

    end

end

            if PlayerAddedConnection then

                PlayerAddedConnection:Disconnect()

                PlayerAddedConnection = nil

            end

            if PlayerRemovingConnection then

                PlayerRemovingConnection:Disconnect()

                PlayerRemovingConnection = nil

            end

            detectedAdmins = {}

        end

    end

local function applyCustomColorToGun(gun, color)

    if not gun or not gun:IsA("Tool") then return end

    if not originalGunColors[gun] then

        originalGunColors[gun] = {}

        for _, part in ipairs(gun:GetDescendants()) do

            if part:IsA("BasePart") and part.Name ~= "Handle" then

                originalGunColors[gun][part] = {

                    Color = part.Color,

                    Material = part.Material

                }

            end

        end

    end

    for _, part in ipairs(gun:GetDescendants()) do

        if part:IsA("BasePart") and part.Name ~= "Handle" then

            part.Color = color

            part.Material = Enum.Material.Neon

        end

    end

end

local function restoreAllGuns()

    for gun, parts in pairs(originalGunColors) do

        for part, data in pairs(parts) do

            if part and part.Parent then

                part.Color = data.Color

                part.Material = data.Material

            end

        end

    end

    originalGunColors = {}

end

CarPanel:AddToggle("Car Speed", false, function(v)

        SpeedEnabled = v

        if not v and Conn then

            Conn:Disconnect()

            Conn = nil

        end

        if v and not Conn then

            Conn = RunService.Heartbeat:Connect(function()

                if not (SpeedEnabled and Seat) then return end

                vel = Seat.AssemblyLinearVelocity

                throttle = Seat.Throttle 

                if Brake or throttle == -1 then

                    Seat.AssemblyLinearVelocity = Vector3.new(0, vel.Y, 0)

                    return

                end

                if Forward or throttle == 1 then

                    dir = Seat.CFrame.LookVector

                    Seat.AssemblyLinearVelocity =

                        Vector3.new(dir.X * SpeedValue, vel.Y, dir.Z * SpeedValue)

                end

                if vel.Y > MaxUpY then

                    Seat.AssemblyLinearVelocity =

                        Vector3.new(

                            Seat.AssemblyLinearVelocity.X,

                            MaxUpY,

                            Seat.AssemblyLinearVelocity.Z

                        )

                end

                Seat.AssemblyLinearVelocity =

                    Seat.AssemblyLinearVelocity + Vector3.new(0, -GroundForce, 0)

            end)

        end

    end

)

local function GetCivCarList()

    local civCars = workspace:FindFirstChild("CivCars")

    local list = {}

    if not civCars then

        return {"Empty"}

    end

    for _, car in ipairs(civCars:GetChildren()) do

        if car:IsA("Model") then

            local seat = car:FindFirstChildWhichIsA("VehicleSeat", true)

            local hasDriver = seat and seat.Occupant ~= nil

            if not hasDriver then

                list[#list + 1] = car.Name

            end

        end

    end

    table.sort(list)

    if #list == 0 then

        return {"Empty"}

    end

    return list

end

local function GetPlayerList()

    local list = {}

    for _, player in ipairs(Players:GetPlayers()) do

        if player ~= LocalPlayer then

            local name = tostring(player.Name)  

            if LocalPlayer:IsFriendsWith(player.UserId) then

                name = name .. " (Friend)"

            end

            table.insert(list, name)

        end

    end

    table.sort(list)

    if #list == 0 then

        return {"Empty"}

    end

    return list

end

local initialList = GetPlayerList()

local function GetClosestPlayerToMouse()

    if not getgenv().SilentAimEnabled then return nil end

    local closestPlayer = nil

    local shortestDistance = math.huge

    local mousePos

    if Mobile then

        local screenSize = Camera.ViewportSize

        mousePos = Vector2.new(screenSize.X / 2, screenSize.Y / 2)

    else

        mousePos = Vector2.new(Mouse.X, Mouse.Y)

    end

    for _, player in ipairs(Players:GetPlayers()) do

        if player ~= LocalPlayer

        and player.Character

        and player.Character:FindFirstChild("Humanoid")

        and player.Character.Humanoid.Health > 0 then

            local part = player.Character:FindFirstChild(getgenv().SilentAimHitPart)

            if part and IsPartVisible(part) then   

                local screenPos, onScreen = Camera:WorldToScreenPoint(part.Position)

                if onScreen then

                    local dist = (Vector2.new(screenPos.X, screenPos.Y) - mousePos).Magnitude

                    if getgenv().SilentAimFOVEnabled and dist > getgenv().SilentAimFOVRadius then

                        continue

                    end

                    if dist < shortestDistance then

                        shortestDistance = dist

                        closestPlayer = player

                    end

                end

            end

        end

    end

    return closestPlayer

end

pcall(function()

    local __namecall

    __namecall = hookmetamethod(game, "__namecall", function(...)

        local args = {...}

        local method = getnamecallmethod() or ""

        if checkcaller() then

            return __namecall(...)

        end

        if not getgenv().SilentAimEnabled or not SilentTarget then

            return __namecall(...)

        end

        if math.random(0,100) > getgenv().SilentAimHitChance then

            return __namecall(...)

        end

        if method == "FireServer" or method == "InvokeServer" then

            if tostring(args[1]) == "MainEvent"

            or tostring(args[1]) == "GunRemote"

            or tostring(args[1]) == "ShootEvent" then

                local hitPart = SilentTarget.Character:FindFirstChild(getgenv().SilentAimHitPart)

                if hitPart then

                    args[2] = hitPart

                    return __namecall(unpack(args))

                end

            end

        end

        if method == "Raycast" and getgenv().WallbangEnabled then

            local hitPart = SilentTarget.Character:FindFirstChild(getgenv().SilentAimHitPart)

            if hitPart then

                local origin = args[2]

                args[3] = (hitPart.Position - origin).Unit * 1000

                return __namecall(unpack(args))

            end

        end

        if method:lower():find("findpartonray") and getgenv().WallbangEnabled then

            local hitPart = SilentTarget.Character:FindFirstChild(getgenv().SilentAimHitPart)

            if hitPart then

                local origin = args[2].Origin

                args[2] = Ray.new(origin, (hitPart.Position - origin).Unit * 9e17)

                return hitPart, hitPart.Position, Vector3.new()

            end

        end

        return __namecall(...)

    end)

end)

local function drawBeam(target)

    if not getgenv().KA_Beam then return end

    local char = LocalPlayer.Character

    if not char then return end

    local myHRP = char:FindFirstChild("HumanoidRootPart")

    local myTorso = char:FindFirstChild("UpperTorso") or char:FindFirstChild("Torso")

    if not myHRP or not myTorso then return end

    local tool = char:FindFirstChildOfClass("Tool")

    local gunHandle = tool and tool:FindFirstChild("Handle")

    local gunMuzzle = gunHandle and gunHandle:FindFirstChild("GunFirePoint")

    local startPart = gunMuzzle or gunHandle or myTorso

    local startPos = startPart.Position

    local lookDirection

    if gunHandle then

        lookDirection = gunHandle.CFrame.LookVector

    else

        lookDirection = myHRP.CFrame.LookVector

    end

    local beamCount = 7

    local spreadAngle = 15  

    for i = 1, beamCount do

        local angle = math.rad((i - (beamCount + 1) / 2) * spreadAngle / beamCount)

        local rotatedDir = CFrame.Angles(0, angle, 0) * lookDirection

        local endPos = startPos + (rotatedDir * 500)  

        local endPart = Instance.new("Part")

        endPart.Size = Vector3.new(0.1, 0.1, 0.1)

        endPart.Position = endPos

        endPart.Anchored = true

        endPart.CanCollide = false

        endPart.Transparency = 1

        endPart.Parent = workspace

        local beam = Instance.new("Beam")

        local att0 = Instance.new("Attachment")

        local att1 = Instance.new("Attachment")

        att0.Parent = startPart

        att1.Parent = endPart

        beam.Attachment0 = att0

        beam.Attachment1 = att1

        beam.Color = ColorSequence.new(Color3.fromRGB(147, 51, 234))  

        beam.Width0 = 0.6

        beam.Width1 = 0.6

        beam.FaceCamera = true

        beam.Transparency = NumberSequence.new(0.1)  

        beam.LightEmission = 1.5  

        beam.LightInfluence = 0

        beam.Texture = "rbxassetid://1084991215"  

        beam.TextureMode = Enum.TextureMode.Wrap

        beam.TextureLength = 2

        beam.Parent = startPart

        task.spawn(function()

            local elapsed = 0

            while beam and beam.Parent and elapsed < 1.5 do

                local pulse = 0.5 + math.sin(elapsed * 10) * 0.1

                beam.Width0 = pulse

                beam.Width1 = pulse

                elapsed = elapsed + 0.05

                task.wait(0.05)

            end

        end)

        task.delay(1.5, function()

            if beam then beam:Destroy() end

            if att0 then att0:Destroy() end

            if att1 then att1:Destroy() end

            if endPart then endPart:Destroy() end

        end)

    end

end

local function expandHitboxes()

    while getgenv().HitboxEnabled do

        for _, player in pairs(Players:GetPlayers()) do

            if player ~= LocalPlayer and player.Character then

                local hitboxPart = player.Character:FindFirstChild(getgenv().HitboxPart)

                if hitboxPart and hitboxPart:IsA("BasePart") then

                    if not hitboxPart:FindFirstChild("HitboxExpanded") then

                        local originalSize = hitboxPart.Size

                        local originalTransparency = hitboxPart.Transparency

                        hitboxPart.Size = Vector3.new(getgenv().HitboxSize, getgenv().HitboxSize, getgenv().HitboxSize)

                        hitboxPart.Transparency = 0.5

                        hitboxPart.CanCollide = false

                        hitboxPart.Massless = true

                        local marker = Instance.new("BoolValue")

                        marker.Name = "HitboxExpanded"

                        marker.Value = true

                        marker.Parent = hitboxPart

                        local sizeValue = Instance.new("Vector3Value")

                        sizeValue.Name = "OriginalSize"

                        sizeValue.Value = originalSize

                        sizeValue.Parent = hitboxPart

                        local transValue = Instance.new("NumberValue")

                        transValue.Name = "OriginalTransparency"

                        transValue.Value = originalTransparency

                        transValue.Parent = hitboxPart

                    end

                end

            end

        end

        task.wait(0.1)

    end

end
